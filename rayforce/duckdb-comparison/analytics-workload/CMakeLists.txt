cmake_minimum_required(VERSION 3.25)
project(analytics_workload LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(engine rayforce duckdb)
  add_library(bench_${engine} SHARED native/bridge.c)
  target_include_directories(bench_${engine} PUBLIC native PRIVATE /opt/${engine}/include /opt/${engine}/src)
  target_link_directories(bench_${engine} PRIVATE /opt/${engine}/lib)
  target_link_libraries(bench_${engine} PRIVATE ${engine})
  set_target_properties(bench_${engine} PROPERTIES BUILD_RPATH /opt/${engine}/lib INSTALL_RPATH /opt/${engine}/lib)
  target_compile_options(bench_${engine} PRIVATE -Wall -Wextra -Werror)
  if(engine STREQUAL rayforce)
    target_compile_definitions(bench_${engine} PRIVATE ENGINE_RAYFORCE)
  endif()
  add_executable(cpp_${engine} cpp/main.cpp)
  target_link_libraries(cpp_${engine} PRIVATE bench_${engine})
  target_compile_options(cpp_${engine} PRIVATE -Wall -Wextra -Werror)
endforeach()
add_executable(rayforce_ipc cpp/ipc.cpp)
target_include_directories(rayforce_ipc PRIVATE /opt/rayforce/include)
target_link_directories(rayforce_ipc PRIVATE /opt/rayforce/lib)
target_link_libraries(rayforce_ipc PRIVATE rayforce)
target_compile_options(rayforce_ipc PRIVATE -Wall -Wextra -Werror)
