cmake_minimum_required(VERSION 3.22)
project(seaweedfs_http LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CURL REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json REQUIRED)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp" MAIN_SHA)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/payload.h" PAYLOAD_SHA)
string(SHA256 HTTP_SOURCE_SHA256 "${MAIN_SHA}:${PAYLOAD_SHA}")
add_executable(http-workload main.cpp)
target_compile_definitions(http-workload PRIVATE HTTP_SOURCE_SHA256="${HTTP_SOURCE_SHA256}")
target_compile_options(http-workload PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(http-workload PRIVATE CURL::libcurl OpenSSL::Crypto nlohmann_json::nlohmann_json)
