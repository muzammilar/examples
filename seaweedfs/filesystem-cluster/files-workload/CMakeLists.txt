cmake_minimum_required(VERSION 3.22)
project(seaweedfs_files LANGUAGES CXX)
set(FILESYSTEM_MAGIC "0x65735546" CACHE STRING "Expected Linux filesystem magic")
set(PAYLOAD_SEED_PREFIX "95513" CACHE STRING "Deterministic payload namespace")
option(SYNC_PUBLICATION_WITH_SYNCFS "Flush the mounted filesystem after publication" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(files-workload main.cpp)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp" WORKLOAD_SOURCE_SHA256)
target_compile_definitions(files-workload PRIVATE WORKLOAD_SOURCE_SHA256="${WORKLOAD_SOURCE_SHA256}")
target_compile_definitions(files-workload PRIVATE EXPECTED_FILESYSTEM_MAGIC=${FILESYSTEM_MAGIC} PAYLOAD_SEED_PREFIX="${PAYLOAD_SEED_PREFIX}" SYNC_PUBLICATION_WITH_SYNCFS=$<BOOL:${SYNC_PUBLICATION_WITH_SYNCFS}>)
target_link_libraries(files-workload PRIVATE OpenSSL::Crypto nlohmann_json::nlohmann_json)
target_compile_options(files-workload PRIVATE -Wall -Wextra -Wpedantic -Werror)
