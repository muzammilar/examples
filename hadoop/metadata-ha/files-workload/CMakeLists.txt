cmake_minimum_required(VERSION 3.22)
project(hdfs_files LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json REQUIRED)
find_path(HDFS_INCLUDE_DIR hdfs.h PATHS /opt/hadoop/include REQUIRED)
find_library(HDFS_LIBRARY hdfs PATHS /opt/hadoop/lib/native REQUIRED)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp" WORKLOAD_SOURCE_SHA256)
add_executable(files-workload main.cpp)
target_compile_definitions(files-workload PRIVATE WORKLOAD_SOURCE_SHA256="${WORKLOAD_SOURCE_SHA256}")
target_include_directories(files-workload PRIVATE ${HDFS_INCLUDE_DIR})
target_link_libraries(files-workload PRIVATE ${HDFS_LIBRARY} OpenSSL::Crypto nlohmann_json::nlohmann_json)
target_compile_options(files-workload PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(files-workload PROPERTIES BUILD_RPATH "/opt/hadoop/lib/native;/opt/java/openjdk/lib/server" INSTALL_RPATH "/opt/hadoop/lib/native;/opt/java/openjdk/lib/server")
