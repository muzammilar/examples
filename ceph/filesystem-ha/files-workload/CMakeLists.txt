cmake_minimum_required(VERSION 3.22)
project(cephfs_files LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json REQUIRED)
find_library(CEPHFS_LIBRARY cephfs REQUIRED)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp" WORKLOAD_SOURCE_SHA256)
add_executable(cephfs-files main.cpp)
target_compile_definitions(cephfs-files PRIVATE _FILE_OFFSET_BITS=64 WORKLOAD_SOURCE_SHA256="${WORKLOAD_SOURCE_SHA256}")
target_link_libraries(cephfs-files PRIVATE ${CEPHFS_LIBRARY} OpenSSL::Crypto nlohmann_json::nlohmann_json)
target_compile_options(cephfs-files PRIVATE -Wall -Wextra -Wpedantic -Werror)
