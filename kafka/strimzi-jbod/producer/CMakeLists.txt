cmake_minimum_required(VERSION 3.20)
project(kafka_order_producer LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(Protobuf REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(RDKAFKA REQUIRED IMPORTED_TARGET rdkafka++=2.12.1)
find_package(nlohmann_json 3.11 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(ZLIB REQUIRED)

protobuf_generate_cpp(PROTO_SOURCES PROTO_HEADERS proto/order.proto)
add_executable(order-producer src/main.cpp ${PROTO_SOURCES} ${PROTO_HEADERS})
target_include_directories(order-producer PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(order-producer PRIVATE protobuf::libprotobuf PkgConfig::RDKAFKA nlohmann_json::nlohmann_json OpenSSL::Crypto ZLIB::ZLIB)
target_compile_options(order-producer PRIVATE -Wall -Wextra -Wpedantic -Werror)

include(CTest)
if(BUILD_TESTING)
  add_executable(witness-tests src/witness_test.cpp)
  target_link_libraries(witness-tests PRIVATE nlohmann_json::nlohmann_json OpenSSL::Crypto ZLIB::ZLIB)
  target_compile_options(witness-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME durable-witness-controls COMMAND witness-tests)
endif()
